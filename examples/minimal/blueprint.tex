\begin{definition}\label{def:even}
  \agda{Data.Nat.Even}\agdaok
A natural number $n$ is \emph{even} if $n = 2k$ for some $k$.
\end{definition}

\begin{lemma}\label{lem:zero-even}\uses{def:even}
  \agda{Data.Nat.Even.zero-even}\agdaok
Zero is even.
\end{lemma}

\begin{theorem}\label{thm:sum-even}\uses{def:even, lem:zero-even}
The sum of two even numbers is even.
\end{theorem}

\begin{proof}\uses{lem:zero-even}
By induction; not yet formalized.
\end{proof}
